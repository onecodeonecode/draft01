'; // <--- PUT YOUR KEY HERE const TICKERS = ['NVDA', 'TSLA', 'AAPL', 'AMD', 'MSFT', 'META', 'AMZN', 'GOOGL', 'NFLX', 'COIN']; async function getStockData(ticker) { try { const res = await fetch(`https://finnhub.io/api/v1/quote?symbol=${ticker}&token=${API_KEY}`); const data = await res.json(); // Return structured data return { symbol: ticker, price: data.c, changePercent: ((data.c - data.pc) / data.pc * 100).toFixed(2) }; } catch (err) { console.error("Error fetching " + ticker, err); return null; } } async function renderApp() { const listContainer = document.getElementById('stock-list'); // Fetch all stocks in parallel const results = await Promise.all(TICKERS.map(t => getStockData(t))); // Filter and Sort by best daily performance const sorted = results .filter(item => item && item.price !== 0) .sort((a, b) => b.changePercent - a.changePercent); listContainer.innerHTML = ''; sorted.forEach(stock => { const isPos = stock.changePercent >= 0; const card = document.createElement('div'); card.className = 'stock-card'; card.innerHTML = `
${stock.symbol} High Volatility Pick
$${stock.price.toLocaleString(undefined, {minimumFractionDigits: 2})}
${isPos ? '▲' : '▼'} ${Math.abs(stock.changePercent)}%
`; listContainer.appendChild(card); }); } // Initial Load renderApp(); // Auto-refresh every 60 seconds setInterval(renderApp, 60000);
